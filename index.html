const WEB_APP_URL = "https://script.google.com/macros/s/AKfycbzfOSbqvB5IsyT0jIl9yPRGsDZfOWJ1xv5XMeBYQWV1K81GjSzjCNzCMOAotTQwGk9lQQ/exec"; function loadStaffList() { const staffStatus = document.getElementById("staffStatus"); staffStatus.innerText = "Loading staff names..."; const oldScript = document.getElementById("staffListScript"); if (oldScript) { oldScript.remove(); } const script = document.createElement("script"); script.id = "staffListScript"; script.src = WEB_APP_URL + "?mode=staffList" + "&callback=staffListCallback" + "&t=" + Date.now(); script.onerror = function() { staffStatus.innerText = "❌ Unable to load Staff List."; }; document.body.appendChild(script); } function staffListCallback(names) { const staffSelect = document.getElementById("staff"); const staffStatus = document.getElementById("staffStatus"); staffSelect.innerHTML = ''; names.forEach(function(name) { const option = document.createElement("option"); option.value = name; option.textContent = name; staffSelect.appendChild(option); }); staffStatus.innerText = "✅ " + names.length + " staff loaded"; } loadStaffList();
